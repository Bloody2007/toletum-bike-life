  <script>
    const CONFIG_TIENDA = {
      emailPedidos: "tuemaildondeLleganLosPedidos@gmail.com",
      numeroBizum: "+34 600 000 000",
      linkPayPal: "https://paypal.me/tuUsuarioPayPal"
    };

    let selectedProduct = { name: '', price: 0 };

    function openCheckout(name, price) {
      selectedProduct = { name: name, price: price };
      document.getElementById('checkoutProductInfo').textContent = `Producto: ${name} - Precio: ${price.toFixed(2)} €`;
      document.getElementById('checkoutModal').style.display = 'flex';
    }

    function closeCheckout() {
      document.getElementById('checkoutModal').style.display = 'none';
    }

    function toggleAddressField() {
      const type = document.getElementById('deliveryType').value;
      const addrGroup = document.getElementById('addressGroup');
      addrGroup.style.display = (type === 'Entrega en Mano (Toledo)') ? 'none' : 'block';
    }

    function processOrder(e) {
      e.preventDefault();
      const name = document.getElementById('custName').value;
      const email = document.getElementById('custEmail').value;
      const delivery = document.getElementById('deliveryType').value;
      const address = document.getElementById('custAddress').value;
      const payment = document.getElementById('paymentMethod').value;

      const bodyText = `NUEVO PEDIDO TOLETUM BIKELIFE%0D%0A----------------------------------%0D%0A` +
        `Producto: ${selectedProduct.name}%0D%0A` +
        `Precio: ${selectedProduct.price.toFixed(2)} €%0D%0A` +
        `Cliente: ${name}%0D%0A` +
        `Email: ${email}%0D%0A` +
        `Tipo de entrega: ${delivery}%0D%0A` +
        `Dirección: ${delivery === 'Envío a Domicilio' ? address : 'Entrega en mano en Toledo'}%0D%0A` +
        `Método de pago elegido: ${payment}%0D%0A`;

      window.location.href = `mailto:${CONFIG_TIENDA.emailPedidos}?subject=Nuevo Pedido - ${encodeURIComponent(name)}&body=${bodyText}`;

      setTimeout(() => {
        if (payment === 'Bizum') {
          alert(`¡Pedido Registrado!\n\nPor favor realiza el Bizum de ${selectedProduct.price.toFixed(2)}€ al número: ${CONFIG_TIENDA.numeroBizum}\nIndica tu nombre (${name}) en el concepto.`);
        } else if (payment === 'PayPal') {
          alert(`¡Pedido Registrado!\n\nSerás redirigido a PayPal para abonar ${selectedProduct.price.toFixed(2)}€.`);
          window.open(CONFIG_TIENDA.linkPayPal, '_blank');
        }
        closeCheckout();
      }, 500);
    }
  </script>
